# 축약 낭독판 (`_ko_audio_brief`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a ~20분 축약 낭독판(`_ko_audio_brief.md`) tier for long papers — condensed from the explainer, additive to the full `_ko_audio.md`, surfaced as the default under the viewer's 듣기 toggle with a "전체" switch, and auto-generated by a length-gated nightly batch.

**Architecture:** Mirror the existing `_ko_audio` plumbing end-to-end. Backend gains a parallel detection flag / path resolver / API route / exclusion guards. The viewer threads a brief-vs-full choice through audio getters via small helpers and skips the mp3 player for the brief (mp3 is phase 2). A new sibling skill `paper-audio-brief-korean` generates the markdown; a length-gated finder + dispatch + Tori cron automate it.

**Tech Stack:** FastAPI + Alpine.js viewer (Docker), Python batch (`main_terminal.py`/skills), pytest, openclaw cron (Tori).

**Spec:** `docs/superpowers/specs/2026-06-06-paper-audio-brief-design.md`

from app import config as _cfg
from app.services import papers, chat


@pytest.fixture(autouse=True)
def _rebind_settings(tmp_workspace, monkeypatch):
    # papers.py AND chat.py both do `from ..config import settings` (module-level
    # binding), so rebind BOTH or chat.load_paper_chunks() won't see tmp_workspace.
    monkeypatch.setattr(papers, "settings", _cfg.settings)

def _make_paper(ws, name, files):
    d = ws / "outputs" / name
    d.mkdir(parents=True, exist_ok=True)
    for fn, content in files.items():
        (d / fn).write_text(content, encoding="utf-8")
    return d


def test_brief_detected_as_its_own_format(tmp_workspace):
    _make_paper(tmp_workspace, "Foo", {
        "Foo.md": "# en",
        "Foo_ko.md": "# ko",
        "Foo_ko_audio.md": "# audio",
        "Foo_ko_audio_brief.md": "# brief",
    })
    info = papers.get_paper_info("Foo")
    f = info["formats"]
    assert f["md_ko_audio_brief"] is True
    assert f["md_ko_audio"] is True   # full audio still detected
    assert f["md_en"] is True         # real English still detected
    assert f["md_ko"] is True


def test_get_md_ko_audio_brief_path_returns_brief(tmp_workspace):
    _make_paper(tmp_workspace, "Foo", {
        "Foo_ko_audio.md": "# audio",
        "Foo_ko_audio_brief.md": "# brief",
    })
    p = papers.get_md_ko_audio_brief_path("Foo")

    assert ok is False  # no real English file exists; brief must not be picked
    assert (tmp_workspace / "outputs" / "Baz" / "Baz_ko_audio_brief.md").read_text() == "# brief"

def test_resolve_result_block_detects_brief(tmp_workspace):

def test_mcp_zip_excludes_brief_when_translation_off(tmp_workspace):
    # mcp_zip gates _ko*.md by include_translation; brief must be gated identically,
    # else _ko_audio_brief.md leaks into the translation-excluded zip as a plain .md.

// Mirror of viewer.html audio brief-vs-full resolution + first-view default.
function decide(p) {
  const {hasMdKoAudioBrief=false, hasMdKoAudio=false, hasMdKoExplained=false,
         hasMdKo=false, hasMdEn=false,
         audioPref=null, explainedPref=null, langPref=null, audioFull=false} = p;
  let langKo = (langPref === 'ko');
  let audioMode = false, explainedMode = false;
  if ((hasMdKoExplained) && explainedPref === 'true') explainedMode = true;
  if (hasMdKoAudio && audioPref === 'true') { audioMode = true; explainedMode = false; langKo = true; }
  if (!audioMode && !explainedMode) {
    const allowKo = langPref !== 'en';
    if (hasMdKoAudio && audioPref === null && allowKo) { audioMode = true; langKo = true; }
    else if (audioPref === null && explainedPref === null && allowKo && hasMdKoExplained) { explainedMode = true; langKo = true; }
    else if (langPref === null && hasMdKo) { langKo = true; }
  }
  const audioUsesBrief = audioMode && hasMdKoAudioBrief && !audioFull;
  if (audioMode) return audioUsesBrief ? 'AUDIO_BRIEF' : 'AUDIO_FULL';
  if (explainedMode) return 'EXPLAINED';
  return langKo ? (hasMdKo ? 'ORIG(ko)' : 'ORIG(en)') : (hasMdEn ? 'ORIG(en)' : 'ORIG(ko)');
}
const T = (d,g,w)=>{const ok=g===w;console.log(`${ok?'✓':'✗'} ${d}: ${g}${ok?'':' WANT '+w}`);if(!ok)process.exitCode=1;};

const FULL = {hasMdKoAudioBrief:true, hasMdKoAudio:true, hasMdKoExplained:true, hasMdKo:true, hasMdEn:true};

T('fresh, full audio only -> full', decide({...FULL, hasMdKoAudioBrief:false}), 'AUDIO_FULL');
T('audioFull switch -> full', decide({...FULL, audioFull:true}), 'AUDIO_FULL');
T('no audio at all -> explained', decide({hasMdKoExplained:true, hasMdKo:true}), 'EXPLAINED');

T('explicit audio on -> brief (default sub-mode)', decide({...FULL, audioPref:'true'}), 'AUDIO_BRIEF');

```python
#!/usr/bin/env python3
"""Find long listen editions (_ko_audio.md) that lack an abridged brief.

A doc qualifies when: a sibling _ko_audio.md exists AND its length exceeds
AUDIO_FULL_MIN_CHARS (~30분) AND a _ko_explained.md source exists AND the
_ko_audio_brief.md is missing OR stale (its sidecar's source_sha256 != the
current explainer). The brief is generated FROM the explainer, so `source`
points at the explainer and `expected_output` at the brief.

Newest-first by the full audio's mtime, capped (default 10). Exits 0 whether or
not anything is missing; only a crash exits non-zero.
"""
import argparse, hashlib, json, os, sys
from pathlib import Path

DEFAULT_ROOT = os.environ.get('PAPERFLOW_OUTPUTS', '/home/restful3/workspace/paperflow/outputs')
DEFAULT_LIMIT = 10
AUDIO_FULL_MIN_CHARS = int(os.environ.get('PAPERFLOW_BRIEF_MIN_CHARS', '10000'))
SKIP_TOKENS = ('_backup_', '.bak', '.tmp')
AUDIO_SUFFIX = '_ko_audio.md'
EXPLAINED_SUFFIX = '_ko_explained.md'
BRIEF_SUFFIX = '_ko_audio_brief.md'
DENYLIST = ('종합건강검진', 'health examination', 'this site can')


def is_skipped(p: Path, root: Path) -> bool:
    try:
        parts = p.relative_to(root).parts
    except ValueError:
        parts = (p.name,)
    return any(tok in part for part in parts for tok in SKIP_TOKENS)


def is_denied(p: Path) -> bool:
    hay = (p.parent.name + '\n' + p.name).lower()
    return any(tok.lower() in hay for tok in DENYLIST)


def is_brief_stale(brief: Path, explained: Path) -> bool:
    """True if an existing brief should be regenerated: sidecar missing/garbled,
    or its recorded source_sha256 differs from the current explainer."""
    sidecar = brief.with_name(brief.name[: -len('.md')] + '.meta.json')
    try:
        meta = json.loads(sidecar.read_text(encoding='utf-8'))
    except (OSError, ValueError):
        return True
    recorded = meta.get('source_sha256')
    if not recorded:
        return True
    try:
        cur = hashlib.sha256(explained.read_bytes()).hexdigest()
    except OSError:
        return False  # can't read source now -> don't churn
    return recorded != cur


def find_missing(root: Path) -> list[dict]:
    items: list[dict] = []
    if not root.is_dir():
        print(f"warning: outputs root not found: {root}", file=sys.stderr)
        return items
    for audio in root.rglob('*' + AUDIO_SUFFIX):
        if not audio.is_file() or is_skipped(audio, root) or is_denied(audio):
            continue
        stem = audio.name[: -len(AUDIO_SUFFIX)]
        explained = audio.with_name(stem + EXPLAINED_SUFFIX)
        brief = audio.with_name(stem + BRIEF_SUFFIX)
        if not explained.exists():
            continue
        # regenerate when brief is missing OR stale vs the current explainer
        if brief.exists() and not is_brief_stale(brief, explained):
            continue
        try:
            n = len(audio.read_text(encoding='utf-8', errors='ignore'))
            mtime = explained.stat().st_mtime   # sort by source (explainer) recency
        except OSError:
            continue
        if n <= AUDIO_FULL_MIN_CHARS:
            continue
        items.append({'folder': str(audio.parent), 'source': str(explained),
                      'expected_output': str(brief), 'full_audio_chars': n, '_mtime': mtime})
    items.sort(key=lambda it: (-it['_mtime'], it['source']))
    for it in items:
        it.pop('_mtime', None)
    return items


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument('--json', action='store_true')
    ap.add_argument('--relative', action='store_true')
    ap.add_argument('--limit', type=int, default=DEFAULT_LIMIT)
    ap.add_argument('--root', default=DEFAULT_ROOT)
    args = ap.parse_args()
    if args.limit < 0:
        ap.error('--limit must be >= 0')
    root = Path(args.root).expanduser()
    missing = find_missing(root)
    if args.limit:
        missing = missing[: args.limit]
    if args.relative:
        base = root.parent
        for item in missing:
            for key in ('folder', 'source', 'expected_output'):
                try:
                    item[key] = str(Path(item[key]).relative_to(base))
                except ValueError:
                    pass
    if args.json:
        print(json.dumps(missing, ensure_ascii=False, indent=2))
    else:
        for item in missing:
            print(item['source'])
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
```
